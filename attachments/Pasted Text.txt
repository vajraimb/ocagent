OCaml 5 + Eio AI Agent Harness (dsh-ocaml) 实施规范
本规范旨在基于 OCaml 5 的代数效应（Algebraic Effects）与 Eio 并发库，实现一套具备“Everything is a Plugin”特性的工业级 AI Agent 运行套件（Harness），全面对标 DeepSeek Harness 架构思想。
1. 整体架构设计
体系分为四层解耦架构：
+-------------------------------------------------------------------+
| 4. Agent 核心层 (Pure Logic)                                       |
|    - 纯逻辑 ReAct 递归循环                                          |
|    - 只发出 Effect，不感知任何具体 I/O / API                        |
+-------------------------------------------------------------------+
                               |  perform Effect
                               v
+-------------------------------------------------------------------+
| 3. Effect 协议接口层 (Core Effect Signatures)                       |
|    - AskLLM | CallTool | SandboxExec | EmitTrajectory | AskApproval  |
+-------------------------------------------------------------------+
                               |  handled by
                               v
+-------------------------------------------------------------------+
| 2. Handler 中间件插件栈 (Plugin Stack / Handlers)                   |
|    - Trajectory Logger | Safety Approval | Budget Gate | LLM Router   |
+-------------------------------------------------------------------+
                               |  executed in
                               v
+-------------------------------------------------------------------+
| 1. Eio 基础设施与能力控制层 (Capability-based Execution)           |
|    - Direct-style HTTP/SSE | FS Capability Sandbox | Fiber Scheduler |
+-------------------------------------------------------------------+

1.1 设计原则
 * 控制反转（IoC）： Agent 逻辑为纯逻辑，任何对外界的依赖（模型、工具、网络、文件系统）均通过 Effect.perform 声明。
 * 能力隔离（Capability-based）： 基于 Eio 的 Capability 范式控制文件与网络权限，沙箱执行天然受限。
 * 组合优于继承： 通过洋葱圈式的 Handler 嵌套组合 Harness 功能，支持零修改无缝切换 Mock 评测（Eval）与生产环境。

3. 核心效应协议与类型规约 (lib/core/)
3.1 基础类型 (types.ml)
type role = System | User | Assistant | ToolResult of string

type message = {
  role : role;
  content : string;
}

type tool_call = {
  id : string;
  name : string;
  args_json : string;
}

type llm_response =
  | TextResponse of string
  | ToolCallResponse of tool_call list

type trajectory_event =
  | LLMStart of message list
  | LLMEnd of llm_response
  | ToolStart of string * string
  | ToolEnd of string * (string, string) result
  | ApprovalRequested of string

3.2 核心代数效应 (effects.ml)
open Effect
open Types

type _ Effect.t +=
  | AskLLM : { messages : message list; model : string } -> llm_response Effect.t
  | CallTool : { name : string; args_json : string } -> (string, string) result Effect.t
  | SandboxExec : { cmd : string; timeout_sec : float } -> (string, string) result Effect.t
  | EmitTrajectory : trajectory_event -> unit Effect.t
  | AskApproval : { action : string; risk_level : string } -> bool Effect.t

4. 基于 Eio 的安全沙箱与能力控制 (lib/harness/sandbox.ml)
利用 Eio 限制系统的文件访问作用域，杜绝路径逃逸。
open Effect
open Effect.Deep
open Core.Effects

let with_eio_sandbox 
    ~(cwd : Eio.Fs.dir Eio.Path.t) 
    ~(mgr : _ Eio.Process.mgr) 
    (f : unit -> 'a) : 'a =
  match_with f ()
    { retc = (fun x -> x);
      exnc = raise;
      effc =
        (fun (type b) (eff : b Effect.t) ->
          match eff with
          | SandboxExec { cmd; timeout_sec } ->
            Some (fun (k : (b, _) continuation) ->
              perform (EmitTrajectory (ToolStart ("sandbox_exec", cmd)));
              let result =
                try
                  (* 基于 Eio 挂载的沙箱目录执行命令 *)
                  let stdout = Eio.Process.parse_out mgr Eio.Buf_read.line
                    [ "sh"; "-c"; cmd ] ~cwd in
                  Ok stdout
                with exn ->
                  Error (Printexc.to_string exn)
              in
              perform (EmitTrajectory (ToolEnd ("sandbox_exec", result)));
              continue k result)
          | _ -> None) }

5. Harness 插件 Handlers 实现 (lib/harness/)
5.1 轨迹与日志记录插件 (trajectory.ml)
open Effect
open Effect.Deep
open Core.Effects

let with_trajectory ~(on_event : Core.Types.trajectory_event -> unit) (f : unit -> 'a) : 'a =
  match_with f ()
    { retc = (fun x -> x);
      exnc = raise;
      effc =
        (fun (type b) (eff : b Effect.t) ->
          match eff with
          | EmitTrajectory ev ->
            Some (fun (k : (b, _) continuation) ->
              on_event ev;
              continue k ())
          | _ -> None) }

let with_risk_approval ~(dangerous_tools : string list) (f : unit -> 'a) : 'a =

        (fun (type b) (eff : b Effect.t) ->

6. Runtime 预设与组装套件 (lib/runtime.ml)
统一调度管理不同的 Handler 栈，提供一键调用的运行预设。
type harness_mode =
  | Production
  | DryRun
  | Evaluation of string (* Trajectory Record/Replay Path *)

let run_harness 
    ~(env : Eio_unix.Stdenv.base) 
    ~(mode : harness_mode) 
    ~(agent_entry : unit -> 'a) : 'a =
  Eio.Switch.run @@ fun _sw ->
  let cwd = Eio.Path.(env#fs / "tmp" / "dsh_workdir") in
  let proc_mgr = env#process_mgr in

  match mode with
  | Production ->
      agent_entry
      |> Harness.Approval.with_risk_approval ~dangerous_tools:["delete_file"; "deploy"][span_6](start_span)[span_6](end_span)
      |> Harness.Trajectory.with_trajectory ~on_event:(fun ev -> Logs.info (fun m -> m "Ev: %a" pp_event ev))[span_7](start_span)[span_7](end_span)
      |> Harness.Sandbox.with_eio_sandbox ~cwd ~mgr:proc_mgr
      |> Harness.Llm_provider.with_deepseek_http ~net:env#net ~model:"deepseek-v4"
  | Evaluation replay_path ->
      agent_entry
      |> Harness.Trajectory.with_trajectory ~on_event:(save_to_disk replay_path)[span_8](start_span)[span_8](end_span)
      |> Harness.Sandbox.with_mock_sandbox
      |> Harness.Llm_provider.with_mock_llm_responses replay_path
  | DryRun ->
      agent_entry
      |> Harness.Approval.with_auto_deny[span_9](start_span)[span_9](end_span)
      |> Harness.Sandbox.with_dry_run_sandbox
      |> Harness.Llm_provider.with_deepseek_http ~net:env#net ~model:"deepseek-v4"

7. 交付里程碑与 Roadmap
| 里程碑 | 周期 | 核心交付物 | 校验标准 |
|---|---|---|---|
| M1: Core Engine | W1-W2 | 协议层 (effects.ml) + Agent 纯逻辑 (agent.ml) + Mock Runtime | 单元测试实现 100% 确定性 Mock 流程推演，无网络依赖 |
| M2: Eio & Direct-style HTTP | W3-W4 | cohttp-eio SSE 流式 Client + 基于 Eio Process 的安全沙箱环境 | 支持 SSE 实时 Chunk 解析与基于 Path 限制的受限 Shell 命令执行 |
| M3: Harness Safety & Trace | W5-W6 | 洋葱圈 Handler 栈组装、轨迹持久化 JSON、CLI 可视化面板 | 拦截高危命令触发 AskApproval，且能够保存/重放 Trajectory 轨迹 |
| M4: Actor Extension | W7-W8 | 扩展基于 Eio.Fiber 的 Actor 消息模型 (Receive/Send/Spawn) | 支持树状 Multi-Agent 协同并发调度及 Supervisor 故障恢复 |
